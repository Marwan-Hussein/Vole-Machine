void Machine::loadProgramFile(string &fileName)
{
    MainUI mainui;
    ifstream f(fileName);
    while (!f.good())
    {
        cout << "not existing file please input again: ";
        cin >> fileName;
        f.open(fileName);
    }
    if (!mainui.isvalid_fileINstructions(fileName))
    {
        f.close();
        cout << "Invalid instructions in this file please input a correct instructions file.\n";
        mainui.inputFileName();
        return;
    }
    cout << fileName << " loaded successfully.\n";
    int address = 0;
    string line;
    while (getline(f, line))
        memory.setcell(address++, line);
    f.close();
}

void Machine::outputState() const
{
    Memory memory;
    cout << "Machine State:\n";
    processor.displayState(); // from cpu
    cout << '\n';
    memory.displayState(); // from memory
    registers.displayState();
    cout << "----------------\n";
}

void Machine::runStepByStep()
{
    cout << "\nRunning program step-by-step.\n";

    while (processor.getProgramCounter() != -2)
    { // Check for HALT (-2)
        processor.fetchInstruction();
        processor.executeNextInstruction();
        outputState();
        cout << "Press Enter to continue to the next step: \n";
        cin.ignore();
    }
    cout << "Execution completed.\n";
}
void Machine::run()
{
    while (processor.getProgramCounter() != -2)
    { // Check for HALT (-2)
        processor.fetchInstruction();
        processor.executeNextInstruction();
    }
}

bool MainUI::isvalid_fileINstructions(string &file)
{
    ifstream f(file);
    string line;
    while (getline(f, line))
    {
        if (line.length() != 4)
            return 0;
        if (line[0] < '1' || line[0] > 'D')
            return 0;
        for (int i = 1; i < 4; ++i)
            if (line[i] < '0' || line[i] > 'F')
                return 0;
        if (line[0] == '4' && line[1] != '0')
            return 0;
        if (line[0] == 'C' && line.substr(1, 3) != "000")
            return 0;
    }
    return 1;
}

#include <vector>
#include <string>
#include <iostream>
#include <iomanip>
#include "Memory.h"
using namespace std;

Memory::Memory()
{
    memory.resize(256);
}

void Memory::setcell(int address, const string &val)
{
    /*if (address == 0)
        cout << "output to screen: " << val << '\n';*/
    if (address >= 0 && address < 256)
    {
        memory[address] = val;
    }
}

string Memory::getcell(int address) const
{
    return (address >= 0 && address < 256) ? memory[address] : "0A";
}

void Memory::displayState() const
{
    cout << "Memory State:\n";
    for (int i = 0; i < memory.size(); ++i)
    {
        if (!memory[i].empty())
        {
            cout << "Address " << setw(2) << setfill('0') << hex << uppercase << i << ": " << memory[i] << '\n';
        }
    }
    cout << dec;
}
